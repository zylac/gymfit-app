RewriteEngine On

# =============================================
# .htaccess untuk InfinityFree
# InfinityFree menggunakan document root /htdocs/
# Tapi Laravel perlu diakses dari folder /public/
# File .htaccess ini di letakkan di /htdocs/ (root)
# =============================================

# 1. Redirect ke public/ untuk semua request
RewriteCond %{REQUEST_URI} !^/public/
RewriteRule ^(.*)$ /public/$1 [L]

# 2. Handle storage files (workaround karena gak bisa symlink)
RewriteRule ^storage/(.*)$ /public/storage/$1 [L]

# 3. Proteksi file sensitif
<FilesMatch "\.(env|env\.example|env\.wasmer|md|gitignore|lock)$">
    Order allow,deny
    Deny from all
</FilesMatch>

<FilesMatch "^(artisan|composer\.(json|lock|phar)|package\.json|vite\.config\.js|tailwind\.config\.js|postcss\.config\.js)$">
    Order allow,deny
    Deny from all
</FilesMatch>

# 4. Laravel sudah punya .htaccess sendiri di public/
#    File ini cuma pengalihan dari root ke public/
